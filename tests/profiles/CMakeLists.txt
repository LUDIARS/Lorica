cmake_minimum_required(VERSION 3.24)
project(profile_regression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CheckLanguage)
check_language(C)
if(CMAKE_C_COMPILER)
    enable_language(C)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/LoricaHardening.cmake")

# The host project may be Debug-only or C++-only. Qualify the fixture's
# optimized C/C++ toolchain before attributing failure to an injected fault.
file(REMOVE "${CMAKE_BINARY_DIR}/skip.txt")
if(NOT CASE STREQUAL "normal" AND CASE)
    lorica_extended_hardening_supported(eligible)
    if(NOT eligible)
        file(WRITE "${CMAKE_BINARY_DIR}/skip.txt" "${eligible_REASON}")
        return()
    endif()
endif()

# Inject one negative capability after inclusion, without depending on the host
# compiler happening to lack this particular flag.
if(CASE STREQUAL "unoptimized")
    set(CMAKE_CXX_FLAGS_RELEASE "-O0")
elseif(CASE STREQUAL "missing_flag")
    set(_LORICA_PROBE_CXX__fstack_clash_protection FALSE)
elseif(CASE STREQUAL "unknown_cpu")
    set(CMAKE_SYSTEM_PROCESSOR "unqualified-cpu")
elseif(CASE STREQUAL "older_libc")
    set(_LORICA_PROBE_CXX_RELEASE_FORTIFY_3 FALSE)
elseif(CASE STREQUAL "old_compiler")
    set(CMAKE_CXX_COMPILER_VERSION 11.4.0)
elseif(CASE STREQUAL "abi32")
    set(CMAKE_SIZEOF_VOID_P 4)
elseif(CASE STREQUAL "no_stdlib")
    set(_LORICA_PROBE_EXTENSIVE_Release_STDLIB FALSE)
elseif(CASE STREQUAL "no_autoinit")
    set(_LORICA_PROBE_CXX__ftrivial_auto_var_init_zero FALSE)
endif()
lorica_extended_hardening_supported(supported)
file(WRITE "${CMAKE_BINARY_DIR}/supported.txt" "${supported}\n${supported_REASON}")
if(REQUIRE_EXTENDED)
    add_executable(strict main.cpp)
    lorica_harden_target(strict EXTENDED)
    return()
endif()

add_executable(control main.cpp)
add_executable(baseline main.cpp)
lorica_harden_target(baseline)
lorica_harden_target(baseline)
target_compile_definitions(baseline PRIVATE LORICA_EXPECT_FAST)
# Baseline libraries: the link options differ by target kind.
add_library(shared SHARED main.cpp)
add_library(static STATIC main.cpp)
foreach(library shared static)
    lorica_harden_target(${library})
    target_compile_definitions(${library} PRIVATE LORICA_EXPECT_FAST)
endforeach()
set(profiles control baseline shared static)
if(supported)
    add_executable(direct main.cpp)
    lorica_harden_target(direct EXTENDED)
    add_executable(upgrade main.cpp)
    lorica_harden_target(upgrade)
    lorica_harden_target(upgrade EXTENDED)
    lorica_harden_target(upgrade EXTENDED)
    lorica_harden_target(upgrade)
    foreach(profile direct upgrade)
        target_compile_definitions(${profile} PRIVATE LORICA_EXPECT_EXTENSIVE
            "$<$<CONFIG:Release,RelWithDebInfo,MinSizeRel>:LORICA_EXPECT_FORTIFY_3>")
    endforeach()
    list(APPEND profiles direct upgrade)
endif()
if(CMAKE_C_COMPILER)
    foreach(profile IN LISTS profiles)
        target_sources(${profile} PRIVATE unit.c)
    endforeach()
endif()
# The link state check.py compares, one line per target.
list(TRANSFORM profiles REPLACE "^(.+)$"
    "\\1|$<TARGET_PROPERTY:\\1,LINK_OPTIONS>|$<TARGET_PROPERTY:\\1,POSITION_INDEPENDENT_CODE>" OUTPUT_VARIABLE link)
list(JOIN link "\n" link)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/link-$<CONFIG>.txt" CONTENT "${link}\n")

# A consumer can qualify a second toolchain flag scope in the same configure.
# Its outcome must not reuse a positive result from the parent scope: neither for
# the Release flags nor for the flags of every configuration.
if(supported AND CASE STREQUAL "normal" AND
   (CMAKE_BUILD_TYPE STREQUAL "Release" OR "Release" IN_LIST CMAKE_CONFIGURATION_TYPES))
    function(check_scope variable value reason)
        set(${variable} "${value}")
        lorica_extended_hardening_supported(local)
        if(local OR NOT local_REASON MATCHES "${reason}")
            message(FATAL_ERROR "Stale qualification with ${variable}=${value}: ${local} ${local_REASON}")
        endif()
    endfunction()
    check_scope(CMAKE_CXX_FLAGS_RELEASE "-O0" "requires effective _FORTIFY_SOURCE=3")
    check_scope(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -florica-no-such-flag" "requires -fstack-clash-protection")
    lorica_extended_hardening_supported(restored)
    if(NOT restored)
        message(FATAL_ERROR "Outer qualification changed: ${restored_REASON}")
    endif()
endif()
